\begin{frame}[fragile]{Metadata JSON: study and dataset context}
  \begin{columns}[T,onlytextwidth]
    \column{0.38\textwidth}
      \begin{block}{Common fields}
        \footnotesize
        \begin{itemize}
          \item Identity: \texttt{dataset\_id}, study title, year and authors
          \item Context: modalities, perturbation, tissue, cell type and cell line
          \item Scale: target and sample counts
          \item Full JSON also has assay, ontology, license and processing fields
        \end{itemize}
      \end{block}
      \footnotesize Repeated annotations and counts are JSON arrays.

  \vfill
  \begin{block}{Record interpretation}

\begin{frame}{Per-dataset artifact names}
  \small
  \footnotesize Paths are relative to the release bucket; an optional version prefix may precede \texttt{perturb-seq/}.
  \vspace{0.3em}
  \begin{tabularx}{\textwidth}{@{}l@{\hspace{1em}}X@{}}
    \toprule
    \textbf{Artifact} & \textbf{Object name pattern} \\
    \midrule
    Metadata & \texttt{perturb-seq/<dataset\_id>.metadata.json} \\
    DEA data & \texttt{perturb-seq/<dataset\_id>.parquet} \\
    DEA CSV mirror & \texttt{perturb-seq/<dataset\_id>.csv.gz} \\
    GSEA data & \texttt{perturb-seq/<dataset\_id>.gsea.parquet} \\
    GSEA CSV mirror & \texttt{perturb-seq/<dataset\_id>.gsea.csv.gz} \\
    \bottomrule
  \end{tabularx}
  \vfill
  \begin{itemize}
    \item One metadata JSON plus DEA and GSEA files for each of the 20 dataset IDs.
    \item Parquet preserves typed columns; CSV.GZ uses the matching human-readable field labels.
    \item The metadata object contains the join key; DEA and GSEA rows are scoped by their per-dataset filenames.
  \end{itemize}
\end{frame}

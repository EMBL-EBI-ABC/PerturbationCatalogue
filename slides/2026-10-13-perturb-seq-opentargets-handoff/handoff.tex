\begin{frame}[fragile]{Dataset metadata: study and experimental context}
  \begin{columns}[T,onlytextwidth]
    \column{0.38\textwidth}
      \begin{block}{Common fields}
        \footnotesize
        \begin{itemize}
          \item Identity: \texttt{dataset\_id}, study title, year and authors

\begin{frame}{Per-dataset release file names}
  \small
  \footnotesize Paths are relative to the release bucket; an optional version prefix may precede \texttt{perturb-seq/}.
  \vspace{0.3em}
  \begin{tabularx}{\textwidth}{@{}l@{\hspace{1em}}X@{}}
    \toprule
    \textbf{File} & \textbf{Object name pattern} \\
    \midrule
    Metadata & \texttt{perturb-seq/<dataset\_id>.metadata.json} \\
    DEA data & \texttt{perturb-seq/<dataset\_id>.parquet} \\
    DEA CSV mirror & \texttt{perturb-seq/<dataset\_id>.csv.gz} \\
    GSEA data & \texttt{perturb-seq/<dataset\_id>.gsea.parquet} \\
    GSEA CSV mirror & \texttt{perturb-seq/<dataset\_id>.gsea.csv.gz} \\
    \bottomrule
  \end{tabularx}
  \vfill
  \begin{itemize}
    \item One metadata JSON plus DEA and GSEA files for each dataset ID.
    \item Parquet preserves typed columns; CSV.GZ uses the matching human-readable field labels.
    \item The metadata object contains the join key; DEA and GSEA rows are scoped by their per-dataset file names.
  \end{itemize}
\end{frame}

\begin{frame}{CRISPR screens: target-level score and interpretation}
  \small
  \begin{tabularx}{\textwidth}{@{}>{\ttfamily}l X@{}}
    \toprule
    \textbf{Export field} & \textbf{Meaning} \\
    \midrule
    perturbed\_target\_ensg & Ensembl gene targeted by the screen \\
    perturbed\_target\_name & Display symbol for the target \\
    score\_name & Name of the assay's score \\
    score\_value & Measured score for the target \\
    significant & Source-provided significance call \\
    significance\_criteria & Rule or threshold used for that call \\
    \bottomrule
  \end{tabularx}
  \vfill
  \begin{block}{Interpretation}
    Scores are assay-specific. The significance flag travels with the source criterion so downstream users can interpret calls in the context of each screen.
  \end{block}
\end{frame}

\begin{frame}{MAVE: variant-level functional scores}
  \small
  \begin{tabularx}{\textwidth}{@{}>{\ttfamily}l X@{}}
    \toprule
    \textbf{Export field} & \textbf{Meaning} \\
    \midrule
    perturbed\_target\_ensg & Ensembl gene associated with the variant \\
    perturbed\_target\_name & Display symbol for the target \\
    perturbation\_name & Variant notation supplied by MaveDB \\
    perturbation\_position & Protein position parsed from the variant notation \\
    perturbation\_aa\_wt & Wild-type amino acid parsed from the notation \\
    perturbation\_aa\_change & Amino-acid change parsed from the notation \\
    score\_name & Name of the assay's score \\
    score\_value & Measured functional score \\
    \bottomrule
  \end{tabularx}
  \vfill
  \begin{block}{Interpretation}
    Variant identity and derived protein-change fields stay alongside the original assay score, making residue-level evidence searchable without replacing the source notation.
  \end{block}
